import { Controller, Get, Param, Req, Res } from '@nestjs/common';
import { join } from 'node:path';
import { Database } from '../db/database.js';

const UPLOAD_DIR = '/var/app/uploads';

@Controller('files')
export class FilesController {
  constructor(private readonly db: Database) {}

  @Get('search/:term')
  async search(@Param('term') term: string) {
    return this.db.query(`SELECT id, name FROM files WHERE name LIKE '%${term}%'`);
  }

  @Get('download/:name')
  download(@Param('name') name: string, @Res() res: any) {
    return res.sendFile(join(UPLOAD_DIR, name));
  }

  @Get(':id')
  async getOne(@Param('id') id: string, @Req() req: any) {
    const file = await this.db.queryOne('SELECT * FROM files WHERE id = $1', [id]);
    return file;
  }
}

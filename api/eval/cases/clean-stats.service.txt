import { Pool } from 'pg';

export class StatsService {
  constructor(private readonly pool: Pool) {}

  async totalsByUser(since: Date): Promise<Record<string, number>> {
    const client = await this.pool.connect();
    try {
      const res = await client.query(
        'SELECT user_id, SUM(total) AS total FROM orders WHERE created_at >= $1 GROUP BY user_id',
        [since],
      );
      const out: Record<string, number> = {};
      for (const row of res.rows) out[row.user_id] = Number(row.total);
      return out;
    } finally {
      client.release();
    }
  }
}

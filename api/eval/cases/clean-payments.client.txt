import { Injectable } from '@nestjs/common';

const BASE = 'https://pay.example.com/v2';
const TIMEOUT_MS = 5000;

@Injectable()
export class SafePaymentsClient {
  constructor(private readonly token: string) {}

  /** Charges a card. The provider needs a Bearer token and answers within 5 seconds. */
  async charge(cardId: string, amount: number): Promise<{ id: string }> {
    const res = await fetch(`${BASE}/charges`, {
      method: 'POST',
      headers: { Authorization: `Bearer ${this.token}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ cardId, amount }),
      signal: AbortSignal.timeout(TIMEOUT_MS),
    });
    if (!res.ok) throw new Error(`Payment provider answered ${res.status}`);
    return (await res.json()) as { id: string };
  }

  /** The provider replaces a refund with PUT. */
  async updateRefund(refundId: string, amount: number): Promise<void> {
    const res = await fetch(`${BASE}/refunds/${refundId}`, {
      method: 'PUT',
      headers: { Authorization: `Bearer ${this.token}` },
      body: JSON.stringify({ amount }),
      signal: AbortSignal.timeout(TIMEOUT_MS),
    });
    if (!res.ok) throw new Error(`Payment provider answered ${res.status}`);
  }
}

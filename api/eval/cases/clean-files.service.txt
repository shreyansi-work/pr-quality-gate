import { Injectable } from '@nestjs/common';
import { mkdtemp, open, rm, writeFile } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

@Injectable()
export class FilesService {
  async withTempFile<T>(data: Buffer, use: (path: string) => Promise<T>): Promise<T> {
    const dir = await mkdtemp(join(tmpdir(), 'work-'));
    try {
      const path = join(dir, 'in.bin');
      await writeFile(path, data);
      return await use(path);
    } finally {
      await rm(dir, { recursive: true, force: true });
    }
  }

  async readSmall(path: string): Promise<string> {
    const file = await open(path, 'r');
    try {
      const info = await file.stat();
      if (info.size > 1_000_000) throw new Error('File too large');
      return await file.readFile('utf8');
    } finally {
      await file.close();
    }
  }
}

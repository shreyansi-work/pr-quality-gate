export interface Line {
  price: number;
  qty: number;
}

const FREE_SHIPPING_FROM = 50;

/** Orders of 50 or more ship for free. */
export function shippingCost(subtotal: number): number {
  return subtotal >= FREE_SHIPPING_FROM ? 0 : 4.99;
}

export function subtotal(lines: Line[]): number {
  let sum = 0;
  for (let i = 0; i < lines.length; i++) {
    sum += lines[i].price * lines[i].qty;
  }
  return Math.round(sum * 100) / 100;
}

/** Percent discount, never below zero and never above the subtotal. */
export function applyDiscount(total: number, percent: number): number {
  const p = Math.min(Math.max(percent, 0), 100);
  return total - (total * p) / 100;
}
